public Class Dashboard{
    public User user;
    private final ExecutorService executor = Executors.newFixedThreadPool(3);
    private final ProfileService profileService;
    private final HistoryService historyService;
    private final RecommendationService recommendationService;

    Dashboard(User user){
        this.user = user;
    }

    public DashboardResponse dashboardResponse(User user){

    CompletableFuture<Result> f1 = CompletableFuture.supplyAsync(() -> profileService.getProfile(user), executor)
                                   .exceptionally(ex->null);
    CompletableFuture<Result> f2 = CompletableFuture.supplyAsync(() -> historyService.getHistory(user), executor)
                                   .exceptionally(ex->null);
    CompletableFuture<Result> f3 = CompletableFuture.supplyAsync(() -> recommendationService.getRecommendation(user), executor)
                                   .exceptionally(ex->null);

    CompletableFuture.allOf(f1,f2,f3).join();

    return new DashboardResponse(f1.join(),f2.join(), f3.join());
    }
    
}
